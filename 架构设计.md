## 1. 技术选型

### 1.1 核心语言

**TypeScript (ESM)**

- 类型安全，适合长期维护
- Ink 4.x 原生支持 ESM，统一模块系统，避免混用 CJS/ESM 的麻烦
- 严格模式开启，尽早暴露错误

```
// tsconfig.json 关键配置
{
  "compilerOptions": {
    "module": "ESNext",
    "moduleResolution": "bundler",
    "target": "ES2022",
    "strict": true
  }
}
```

### 1.2 TUI 框架

**Ink 4.4 + React 18**

| 优点 | 说明 |
|---|---|
| 声明式 UI | 用 React 组件管理界面状态，天然适合响应式数据 |
| 丰富生态 | ink-spinner、ink-select-input 等开箱即用 |
| 响应式布局 | useStdoutDimensions 自适应终端尺寸 |
| 输入处理 | useInput hook 统一处理快捷键 |

### 1.3 数据库

**sql.js（WebAssembly 版 SQLite）**

> ⚠️ 不选 `better-sqlite3` 的原因：
> `better-sqlite3` 需要在用户机器上编译 C++ 原生模块（依赖 `node-gyp`、Python、MSVC），
> `npm install -g xian` 场景下极容易在 Windows/Linux 上失败。

`sql.js` 是纯 WebAssembly，安装零编译，跨平台稳定。

### 1.4 颜色输出

**picocolors** 替代 chalk

- `chalk@4` 是 CJS，`chalk@5+` 是 ESM，版本混乱
- `picocolors` 轻量（无依赖），与 Ink 无冲突

### 1.5 系统感知

**simple-git** — 读取 Git 活动（无需用户安装任何额外工具）

> **设计决策**：放弃进程监控（systeminformation）与 CPU 监控。  
> 原因：安装复杂度高、跨平台兼容性差、用户体验不稳定。  
> 替代方案：以 Git commit 行为作为唯一现实信号驱动游戏；阵营路径改为玩家在入道仪式中主动声明，永久绑定。

```typescript
import { render } from 'ink';
import { App } from './ui/App.js';
import { initDb } from './storage/index.js';
import { OSync } from './osync/index.js';
import { GameEngine } from './engine/index.js';
import { AiContentLayer } from './ai/index.js';
import { loadConfig } from './config/index.js';

async function main() {
  const config = loadConfig();
  await initDb();

  const osync = new OSync(config);
  const engine = new GameEngine();
  const aiLayer = new AiContentLayer(config);  // 无 key 时静默禁用

  osync.on('signal', (s) => engine.process(s));
  osync.start();
  aiLayer.startBackgroundRefill();             // 后台静默预生成

  const { unmount } = render(<App engine={engine} aiLayer={aiLayer} />);

  process.on('SIGINT', async () => {
    await engine.save();
    osync.stop();
    aiLayer.stop();
    unmount();
    process.exit(0);
  });
}

main();
```

### 4.2 感知层 (OSync) — Git-only

OSync 只轮询 Git 活动，产生归一化的 `RawSignal`。  
阵营路径（orthodox / vibe / unbound）由玩家在入道仪式中声明，不再通过进程检测推断。

#### 信号类型定义

```typescript
// src/osync/signal.types.ts

export interface RawSignal {
  timestamp: number;
  newCommit: CommitInfo | null;
  hour: number;                        // 当前小时，用于时段加成
  minutesSinceLastCommit: number;      // 距上次 commit 的分钟数
}

export interface CommitInfo {
  hash: string;
  message: string;
  insertions: number;
  deletions: number;
}
```

#### 轮询调度

```typescript
// src/osync/index.ts

export class OSync extends EventEmitter {
  private timer: NodeJS.Timeout | null = null;
  private lastSignalTime = Date.now();

  constructor(private config: XianConfig) {
    super();
  }

  start() {
    // Git 活动：每 10 秒
    this.timer = setInterval(() => this.poll(), 10_000);
    // 立即执行一次
    this.poll();
  }

  private async poll() {
    const commit = await this.gitWatcher.poll(this.config.osync.watchDir);
    const now = Date.now();
    const signal: RawSignal = {
      timestamp: now,
      newCommit: commit,
      hour: new Date().getHours(),
      minutesSinceLastCommit: Math.floor((now - this.lastSignalTime) / 60_000),
    };
    if (commit) this.lastSignalTime = now;
    this.emit('signal', signal);
  }

  stop() {
    if (this.timer) clearInterval(this.timer);
  }
}
```

#### 数值计算（Git-based）

```typescript
// src/engine/stat-calculator.ts

export function calcStatDelta(
  signal: RawSignal,
  state: PlayerState
): StatDelta {
  const delta: StatDelta = {
    cultivation: 0,
    foundation: 0,
    vibe: 0,
    heart: 0,
    prestige: 0,
  };

  // ── commit 奖励 ──────────────────────────────────────────
  if (signal.newCommit) {
    const lines = signal.newCommit.insertions + signal.newCommit.deletions;

    if (lines < 50) {
      // 小提交：功德刻印
      delta.cultivation += 5;
      delta.heart       += 1;
    } else if (lines < 300) {
      // 中提交：大道参悟
      delta.cultivation += 20;
      delta.foundation  += 2;
    } else {
      // 大提交：灵脉贯通
      delta.cultivation += 50;
      delta.prestige    += 10;
    }

    // 路径加成（入道时声明，永久生效）
    switch (state.activePath) {
      case 'orthodox':
        delta.foundation += 3;
        delta.heart      += 1;
        break;
      case 'vibe':
        delta.vibe        += 5;
        delta.cultivation += 5;
        break;
      case 'unbound':
        delta.foundation += 2;
        delta.vibe       += 2;
        break;
    }

    // 夜修加成 (22:00 - 06:00)
    if (signal.hour >= 22 || signal.hour <= 5) {
      delta.foundation  *= 1.15;
      delta.cultivation *= 1.15;
    }

    delta.prestige += 5;
  }

  // ── 冥想摸鱼：长时间无提交，道心缓慢恢复 ────────────────
  if (signal.minutesSinceLastCommit > 60) {
    delta.heart += 0.5;
  }

  return clampDelta(delta, state);
}
```

```typescript
// src/narrative/template-engine.ts

export function renderTemplate(
  template: string,
  vars: Record<string, string | number>
): string {
  return template.replace(/\{(\w+)\}/g, (_, key) =>
    String(vars[key] ?? `{${key}}`)
  );
}

export function pickEventText(
  sizeKey: 'small' | 'medium' | 'large',
  faction: FactionKey,
  vars: Record<string, string | number>
): string {
  const pool = COMMIT_TEMPLATES[sizeKey][faction] ?? COMMIT_TEMPLATES[sizeKey].orthodox;
  const raw  = pool[Math.floor(Math.random() * pool.length)];
  return renderTemplate(raw, vars);
}
```

### 4.5 AI 内容生成层

AI 层完全可选，无 API Key 时静默禁用，游戏全程使用静态模板正常运行。

#### 核心设计原则

- **不阻塞主循环**：AI 请求全部异步，结果写入本地缓存池
- **优先消费缓存**：叙事层优先从 AI 内容池取文案，池空时降级静态模板
- **结构化输出**：要求模型返回 JSON，包含事件类型、文案、数值效果

#### 内容池管理

```typescript
// src/ai/content-cache.ts

export class ContentCache {
  private pool: AiEvent[] = [];
  private readonly MIN_POOL_SIZE = 3;
  private readonly MAX_POOL_SIZE = 20;

  hasContent(): boolean {
    return this.pool.length > 0;
  }

  consume(): AiEvent | null {
    return this.pool.shift() ?? null;
  }

  push(events: AiEvent[]) {
    this.pool.push(...events);
    if (this.pool.length > this.MAX_POOL_SIZE) {
      this.pool = this.pool.slice(-this.MAX_POOL_SIZE);
    }
  }

  needsRefill(): boolean {
    return this.pool.length < this.MIN_POOL_SIZE;
  }
}
```

#### 后台刷新调度

```typescript
// src/ai/generator.ts

export class AiGenerator {
  private running = false;

  constructor(
    private cache: ContentCache,
    private config: XianConfig
  ) {}

  async refill(state: PlayerState, recentCommits: CommitInfo[]) {
    if (this.running || !this.config.ai.enabled || !this.config.ai.apiKey) return;
    this.running = true;
    try {
      const prompt = buildEventPrompt(state, recentCommits);
      const raw    = await callAiApi(prompt, this.config.ai);
      const events = JSON.parse(raw) as AiEvent[];
      this.cache.push(events);
    } catch {
      // 静默失败，降级到静态模板
    } finally {
      this.running = false;
    }
  }
}
```

```typescript
// src/storage/db.ts

import initSqlJs, { Database } from 'sql.js';
import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'fs';
import { DB_PATH, APP_DIR } from '../shared/constants.js';
import { INIT_SQL } from './migrations/001_init.js';

let db: Database | null = null;

export async function initDb(): Promise<Database> {
  if (db)
 return db;

  if (!existsSync(APP_DIR)) mkdirSync(APP_DIR, { recursive: true });

  const SQL = await initSqlJs();

  db = existsSync(DB_PATH)
    ? new SQL.Database(readFileSync(DB_PATH))
    : new SQL.Database();

  db.run(INIT_SQL);
  return db;
}

export function getDb(): Database {
  if (!db) throw new Error('DB not initialized');
  return db;
}

export function persistDb(): void {
  if (!db) return;
  writeFileSync(DB_PATH, Buffer.from(db.export()));
}
```

### 6.1 Boss Key（隐匿术）

```typescript
// src/ui/hooks/useBossKey.ts

export function useBossKey(onHide: () => void, onShow: () => void) {
  const [hidden, setHidden] = useState(false);

  useInput((input, key) => {
    if (key.ctrl && input.toLowerCase() === 'b') {
      if (hidden) { onShow(); setHidden(false); }
      else        { onHide(); setHidden(true);  }
    }
  });
}
```

### 6.2 Git 活动感知

> 核心原则：只读，不写，不追踪内容。

```typescript
// src/osync/git-watcher.ts

import simpleGit from 'simple-git';

export class GitWatcher {
  private lastHash: string = '';

  async poll(cwd: string): Promise<CommitInfo | null> {
    try {
      const git    = simpleGit(cwd);
      const log    = await git.log({ maxCount: 1 });
      const latest = log.latest;

      if (!latest || latest.hash === this.lastHash) return null;

      const diff = await git.diffSummary([`${this.lastHash}..HEAD`]);
      this.lastHash = latest.hash;

      return {
        hash:       latest.hash,
        message:    latest.message,
        insertions: diff.insertions,
        deletions:  diff.deletions,
      };
    } catch {
      return null; // 非 git 仓库，静默忽略
    }
  }
}
```

```typescript
// src/engine/offline-calculator.ts

export function calcOfflineProgress(
  lastSaved: number,
  state: PlayerState,
  config: OfflineConfig
): StatDelta {
  const offlineMs   = Date.now() - lastSaved;
  const offlineMins = Math.min(offlineMs / 60_000, config.offlineMaxHours * 60);

  return {
    cultivation: offlineMins * 0.1,   // 离线修为，极低速率
    heart:       offlineMins * 0.3,   // 道心缓慢恢复
    foundation:  0,                   // 离线不涨根基
    vibe:        0,
    prestige:    0,
  };
}
```

## 9. 跨平台策略

| 问题 | Windows | macOS | Linux | 解决方案 |

| 原生模块 | 无（已规避）| 无 | 无 | sql.js (WASM) + 无 systeminformation |
